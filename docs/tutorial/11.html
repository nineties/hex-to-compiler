11. vasm: 所有と借用の検査

use("vasm") は、機械命令に近い低級な言語 vasm を vasm{| ... |} の中に書けるようにします。vasm のプログラムは関数と構造体の並びで、検査器が所有と借用の違反・解放後の使用・未初期化の読み出しを型の規則だけで拒みます。

検査の結果は vasm_errors(file, m) が返す VasmErr{kind, loc, msg} のリストです。誤りのない関数は結果に現れず、1つの関数の誤りは最初の1件だけです。この章の例は、次の report で結果を1行ずつ書きます。

所有と借用

vasm_box.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn bump[r](p: &mut r i64) {
        var v: i64
        v = *p
        v = v + 1
        *p = v
    }
    fn main() -> i64 {
        var p: own i64
        var v: i64
        p = box(41)
        region r {
            var m: &mut r i64
            m = &mut r p
            bump(m)
        }
        v = *p
        free(p)
        return v
    }
|}
report(m)

出力

ok

own i64 は box で確保した整数への、ただ1つの持ち主です。region r { ... } は借用の有効な範囲に名前を付け、&mut r p はその中だけ p を書き換え可能に借ります。bump の [r] は呼び出しごとに決まる領域の引数で、&mut を渡しても呼び出しの間だけ借り直されるので、m はそのあとも使えます。

解放後の使用

vasm_freed.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn use_after_free() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        free(p)
        v = *p
        return v
    }
    fn free_twice() {
        var p: own i64
        p = box(1)
        free(p)
        free(p)
    }
    fn moved_then_used(q: own i64) -> own i64 {
        var p: own i64
        p = q
        free(q)
        return p
    }
|}
report(m)

出力

line 15: p is used after it is freed
line 22: p is freed twice
line 27: q is used after it is moved

free した own は読めず、もう一度 free もできません。q を p に代入すると所有が p に移る（move）ので、q はそのあと使えません。

自動の解放

vasm_drop.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Buf { data: own [u8], size: i64 }
    fn at_block_end(c: bool) {
        if (c) {
            var q: own i64
            q = box(2)
        }
    }
    fn overwrite() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        p = box(2)
        box(3)
        v = *p
        return v
    }
    fn free_with_contents(b: own Buf) {
        free(b)
    }
|}
report(m)

出力

ok

own を持った値は、持ち主がなくなる所で自動的に解放されます。ブロックや関数の終わり、上書き、文として捨てた関数の結果、free した値の中に残ったものが、その場所です。上の例はどれも誤りになりません。free は早く手放したいときに書きます。

借用の規則

vasm_borrow.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn two_shared() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            var b: &r i64
            a = &r p
            b = &r p
            var w: i64
            v = *a
            w = *b
            v = v + w
        }
        free(p)
        return v
    }
    fn mut_and_shared() {
        var p: own i64
        p = box(1)
        region r {
            var a: &mut r i64
            var b: &r i64
            a = &mut r p
            b = &r p
        }
    }
    fn overwrite_while_borrowed() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            a = &r p
            p = box(2)
            v = *a
        }
        return v
    }
|}
report(m)

出力

line 34: p is borrowed as mut and cannot be borrowed
line 44: p is borrowed and cannot be assigned

共有の借用 &r は同時にいくつでも持てます。&mut r は1つだけで、借りている間は元の値を読むことも、書き換えることも、解放することもできません。

領域の内と外

vasm_escape.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn inner_in_outer() {
        var p: own i64
        p = box(1)
        region outer {
            var keep: &outer i64
            region inner {
                var a: &inner i64
                a = &inner p
                keep = a
            }
        }
    }
    fn outer_in_inner() {
        var p: own i64
        p = box(1)
        region outer {
            var a: &outer i64
            a = &outer p
            region inner {
                var b: &inner i64
                b = a
            }
        }
    }
|}
report(m)

出力

line 18: a borrow of region inner is put in keep, which lives outside the region

内側の領域の借用を外側のレジスタに入れると、領域が終わったあとに借用だけが残ります。逆に、外側の領域の借用を内側の領域の借用として使うことは、外のほうが長く生きるので許されます。

整数の幅

vasm_ints.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn mixed(flags: u8, size: u32) -> u32 {
        var w: u32
        w = size + flags
        return w
    }
    fn widened(flags: u8, size: u32) -> u32 {
        var f: u32
        var w: u32
        f = flags as u32
        w = size + f
        return w
    }
    fn magic() -> u32 {
        var m: u32
        m = 0x7F45
        m = m << 16
        m = m | 0x4C46
        return m
    }
    fn too_big() -> u32 {
        var m: u32
        m = 0x7F454C46
        return m
    }
|}
report(m)

出力

line 12: the operands are u32 and u8
line 31: 0x7F454C46 is out of the range of a literal, -32768 to 32767; make it with << and |

整数の型は i8 から i64、u8 から u64 で、演算の両辺は同じ型です。幅や符号を変えるには as を使います。リテラルは -32768 から 32767 までなので、大きな値はシフトと | で組み立てます。

配列と構造体

vasm_array.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Section { offset: u64, size: u32 }
    fn total[r](ss: &r [Section]) -> u64 {
        var i: i64
        var n: i64
        var t: u64
        var s: Section
        var o: u64
        i = 0
        t = 0
        n = len(ss)
        while (i < n) {
            s = ss[i]
            o = s.offset
            t = t + o
            i = i + 1
        }
        return t
    }
    fn build() -> u64 {
        var a: own [Section]
        var s: Section
        var t: u64
        s = Section{offset: 64, size: 16}
        a = alloc(3, s)
        s = Section{offset: 128, size: 0}
        a[1] = s
        region r {
            var view: &r [Section]
            view = &r a
            t = total(view)
        }
        free(a)
        return t
    }
|}
report(m)

出力

ok

alloc(n, v) は v を n 個並べた own [T] を作り、len は長さを返します。添字は i64 で、範囲の検査は実行時に行う前提です。構造体は全部のフィールドを1回ずつ書いて作り、own を持たなければ整数と同じくコピーされます。

own を持つ値の入れ替え

vasm_swap.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Image { header: own [u8], body: own [u8], entry: u64 }
    fn replace_body(img: Image, n: i64) -> Image {
        var b: own [u8]
        b = alloc(n, 0)
        img.body = b
        return img
    }
    fn take_header[r](parts: &mut r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = alloc(1, 9)
        a = swap(parts[0], a)
        return a
    }
    fn read_header[r](parts: &r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = parts[0]
        return a
    }
|}
report(m)

出力

line 25: the elements of parts own values and are not copied; use swap(parts[i], v)

own を持つ値は、配列の要素やポインタの指す先から読んでコピーすることができません（持ち主が2人になるため）。swap(p[i], v) は v を入れて前の値を取り出します。構造体のレジスタのフィールドへの書き込み img.body = b は、前の値を解放して入れ替えます。

規則の全体はリファレンスの vasmにあります。
