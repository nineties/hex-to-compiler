module bitset {
    export new(n) := i32_array(n / 30 + 1)
    helper(i) := i / 30
    export add(s, i) := {
        w := helper(i)
        s[w] = bit_or(s[w], bit_shl(1, i % 30))
    }
    test {
        assert(helper(65) == 2, "helper")
        s := new(100)
        add(s, 3)
        assert(s[0] == 8, "add")
    }
}
