# check_with(m, imports) checks m, whose functions may call the functions and use the structs of imports, a list of # structs and functions (the bodies of the functions are not read). Only the items of m are checked. export check_with(m, imports) := { items := arg(m, 0) sigs = table.new() structs = table.new() each(d -> match d { Fn{_, f, rs, ps, ret, _} -> table.set(sigs, f, Sig{rs, map(p -> arg(p, 2), ps), ret}) | Struct{_, s, _} -> if (table.get(structs, s, none) == none) table.set(structs, s, d) }, append(imports, items)) errs := [] each(d -> { r := handle { if (head(d) == 'Struct) check_struct(d) else check_fn(d) none } with (VasmFail{at, kind, msg}, k) -> VasmErr{kind, at, msg} if (r != none) errs = r :: errs }, items) reverse(errs) } # the errors of the module m: [VasmErr{kind, Pos{d}, msg}, ...], the first error of each struct and function export check(m) := check_with(m, [])
