# the errors of the module m: [VasmErr{kind, Pos{d}, msg}, ...], the first error of each struct and function vasm_check(m) := { items := arg(m, 0) vasm_sigs = table() vasm_structs = table() each(d -> match d { Fn{_, f, rs, ps, ret, _} -> table_set(vasm_sigs, f, Sig{rs, map(p -> arg(p, 2), ps), ret}) | Struct{_, s, _} -> if (table_get(vasm_structs, s, none) == none) table_set(vasm_structs, s, d) }, items) errs := [] each(d -> { r := handle { if (head(d) == 'Struct) vasm_check_struct(d) else vasm_check_fn(d) none } with (VasmFail{at, kind, msg}, k) -> VasmErr{kind, at, msg} if (r != none) errs = r :: errs }, items) reverse(errs) }
