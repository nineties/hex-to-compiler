# a value of type t can be put where e is expected: an integer literal in the range of e, # or a borrow of a region that outlives the region of e vasm_fits(at, e, t) := if (head(t) == 'Lit and int(e)) { vasm_lit_range(at, arg(t, 0), e) true } else if (head(e) == 'Own and head(t) == 'Own) vasm_fits(at, arg(e, 0), arg(t, 0)) else if (head(e) == 'Arr and head(t) == 'Arr) vasm_fits(at, arg(e, 0), arg(t, 0)) else if (head(e) == 'Ref and head(t) == 'Ref) arg(e, 0) == arg(t, 0) and vasm_outlives(arg(t, 1), arg(e, 1)) and vasm_fits(at, arg(e, 2), arg(t, 2)) else e == t # a borrow that does not fit only because its region is opened inside the function and does not outlive the expected one vasm_escapes(e, t) := head(e) == 'Ref and head(t) == 'Ref and arg(e, 0) == arg(t, 0) and arg(e, 2) == arg(t, 2) and not member(arg(t, 1), vasm_fn_regions)
