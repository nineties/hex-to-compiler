fun walk walk = (Leaf{}) -> none | (Node{l, v, r}) -> { walk(l); perform Yield{v}; walk(r) } tree := Node{Node{Leaf{}, 1, Leaf{}}, 2, Node{Leaf{}, 3, Leaf{}}} handle walk(tree) with (Yield{v}, k) -> { put(f"yield {v}\n"); k(none) } to_list(t) := handle { walk(t); [] } with (Yield{v}, k) -> v :: k(none) use("pretty") put(f"{to_list(tree)}\n")
