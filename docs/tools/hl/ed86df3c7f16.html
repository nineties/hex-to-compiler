# A rest parameter x... takes the remaining arguments as a Tuple sum_tuple(t) := { s := 0 i := 0 while (i < arity(t)) { s = s + arg(t, i) i = i + 1 } s } # the last parameter of a function total(xs...) := sum_tuple(xs) println(total()) println(total(1, 2, 3)) # after the fixed parameters scale(k, xs...) := k * sum_tuple(xs) println(scale(10, 1, 2, 3)) # in a function written with -> count := (first, others...) -> 1 + arity(others) println(count(7, 8, 9)) # arms with fewer parameters are chosen by the number of arguments describe(a) := "one" describe(a, b) := "two" describe(a, b, c...) := "three or more" println(describe(1)) println(describe(1, 2)) println(describe(1, 2, 3, 4)) # in the pattern of an M-expression, the arguments after the fixed ones args_of(Call{f, rest...}) := rest println(args_of(Call{1, 2, 3})) println(args_of(Call{1})) # in match println(match Call{1, 2, 3} { Call{x, ys...} -> ys })
