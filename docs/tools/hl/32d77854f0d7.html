vasm_stmt ::= at:vasm_here 'var x:vasm_name ":" t:vasm_type { Decl{at, x, t} } | at:vasm_here 'if "(" c:vasm_cond ")" a:vasm_block 'else b:vasm_block { IfS{at, c, a, b} } | at:vasm_here 'if "(" c:vasm_cond ")" a:vasm_block { IfS{at, c, a, []} } | at:vasm_here 'while "(" c:vasm_cond ")" a:vasm_block { WhileS{at, c, a} } | at:vasm_here 'region r:vasm_name a:vasm_block { RegionS{at, r, a} } | at:vasm_here 'return vasm_same a:vasm_operand { Ret{at, a} } | at:vasm_here 'return { Ret{at, none} } | at:vasm_here "*" p:vasm_name "=" a:vasm_operand { Store{at, p, a} } | at:vasm_here p:vasm_name "[" i:vasm_operand "]" "=" a:vasm_operand { IStore{at, p, i, a} } | at:vasm_here p:vasm_name "." f:vasm_name "=" a:vasm_operand { FStore{at, p, f, a} } | at:vasm_here x:vasm_name "=" r:vasm_rhs { Assign{at, x, r} } | at:vasm_here f:vasm_name "(" args:vasm_args? ")" { CallS{at, f, vasm_opt_list(args)} }
