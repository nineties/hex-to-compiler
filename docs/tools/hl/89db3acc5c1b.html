# q outlives r: q is r, or q is static (the region of the string literals, which lasts to the end of the program), # or r is a region opened in the function and q was opened before it. The regions of the function are not ordered. outlives(q, r) := q == r or q == 'static or (r != 'static and not member(r, fn_regions) and list.index_of(q, regions) > list.index_of(r, regions))
