11. vasm: checking ownership and borrows

use("vasm") lets you write vasm, a low-level language close to machine instructions, inside vasm{| ... |}. A vasm program is a list of functions and structs. The checker rejects ownership and borrow violations, uses after free and reads of uninitialized registers using type rules alone.

The result of the check is the list of VasmErr{kind, loc, msg} that vasm_errors(file, m) returns. A function without an error does not appear in it, and a function has at most one error: the first. The examples in this chapter write the result line by line with the report function in each file.

Ownership and borrows

vasm_box.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn bump[r](p: &mut r i64) {
        var v: i64
        v = *p
        v = v + 1
        *p = v
    }
    fn main() -> i64 {
        var p: own i64
        var v: i64
        p = box(41)
        region r {
            var m: &mut r i64
            m = &mut r p
            bump(m)
        }
        v = *p
        free(p)
        return v
    }
|}
report(m)

Output

ok

own i64 is the single owner of an integer allocated by box. region r { ... } names the extent in which a borrow is valid, and &mut r p borrows p for writing inside it. The [r] of bump is a region parameter fixed at each call. A &mut passed to a call is borrowed again for the duration of the call, so m can be used after it.

Use after free

vasm_freed.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn use_after_free() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        free(p)
        v = *p
        return v
    }
    fn free_twice() {
        var p: own i64
        p = box(1)
        free(p)
        free(p)
    }
    fn moved_then_used(q: own i64) -> own i64 {
        var p: own i64
        p = q
        free(q)
        return p
    }
|}
report(m)

Output

line 15: p is used after it is freed
line 22: p is freed twice
line 27: q is used after it is moved

An own that has been freed cannot be read or freed again. Assigning q to p moves the ownership to p, so q cannot be used afterwards.

Automatic release

vasm_drop.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Buf { data: own [u8], size: i64 }
    fn at_block_end(c: bool) {
        if (c) {
            var q: own i64
            q = box(2)
        }
    }
    fn overwrite() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        p = box(2)
        box(3)
        v = *p
        return v
    }
    fn free_with_contents(b: own Buf) {
        free(b)
    }
|}
report(m)

Output

ok

A value that owns something is released automatically where its owner goes away: at the end of a block or a function, when it is overwritten, when the result of a call is discarded as a statement, and for what is left inside a value that is passed to free. None of the functions above is an error. free is for releasing a value earlier.

Rules for borrows

vasm_borrow.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn two_shared() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            var b: &r i64
            a = &r p
            b = &r p
            var w: i64
            v = *a
            w = *b
            v = v + w
        }
        free(p)
        return v
    }
    fn mut_and_shared() {
        var p: own i64
        p = box(1)
        region r {
            var a: &mut r i64
            var b: &r i64
            a = &mut r p
            b = &r p
        }
    }
    fn overwrite_while_borrowed() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            a = &r p
            p = box(2)
            v = *a
        }
        return v
    }
|}
report(m)

Output

line 34: p is borrowed as mut and cannot be borrowed
line 44: p is borrowed and cannot be assigned

Any number of shared borrows &r can exist at once. A &mut r is exclusive: while it lives, the original value can be neither read, written nor freed.

Inside and outside of regions

vasm_escape.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn inner_in_outer() {
        var p: own i64
        p = box(1)
        region outer {
            var keep: &outer i64
            region inner {
                var a: &inner i64
                a = &inner p
                keep = a
            }
        }
    }
    fn outer_in_inner() {
        var p: own i64
        p = box(1)
        region outer {
            var a: &outer i64
            a = &outer p
            region inner {
                var b: &inner i64
                b = a
            }
        }
    }
|}
report(m)

Output

line 18: a borrow of region inner is put in keep, which lives outside the region

A borrow of an inner region put in an outer register would outlive its region. A borrow of an outer region can be used where a borrow of an inner region is expected, because the outer one lives longer.

Widths of integers

vasm_ints.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn mixed(flags: u8, size: u32) -> u32 {
        var w: u32
        w = size + flags
        return w
    }
    fn widened(flags: u8, size: u32) -> u32 {
        var f: u32
        var w: u32
        f = flags as u32
        w = size + f
        return w
    }
    fn magic() -> u32 {
        var m: u32
        m = 0x7F45
        m = m << 16
        m = m | 0x4C46
        return m
    }
    fn too_big() -> u32 {
        var m: u32
        m = 0x7F454C46
        return m
    }
|}
report(m)

Output

line 12: the operands are u32 and u8
line 31: 0x7F454C46 is out of the range of a literal, -32768 to 32767; make it with << and |

The integer types are i8 to i64 and u8 to u64, and both operands of an operator have the same type. Use as to change the width or the signedness. A literal ranges from -32768 to 32767, so a larger value is built with shifts and |.

Arrays and structs

vasm_array.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Section { offset: u64, size: u32 }
    fn total[r](ss: &r [Section]) -> u64 {
        var i: i64
        var n: i64
        var t: u64
        var s: Section
        var o: u64
        i = 0
        t = 0
        n = len(ss)
        while (i < n) {
            s = ss[i]
            o = s.offset
            t = t + o
            i = i + 1
        }
        return t
    }
    fn build() -> u64 {
        var a: own [Section]
        var s: Section
        var t: u64
        s = Section{offset: 64, size: 16}
        a = alloc(3, s)
        s = Section{offset: 128, size: 0}
        a[1] = s
        region r {
            var view: &r [Section]
            view = &r a
            t = total(view)
        }
        free(a)
        return t
    }
|}
report(m)

Output

ok

alloc(n, v) makes an own [T] of n copies of v, and len returns the length. An index is an i64, and checking its range is left to run time. A struct is built by giving every field once. It is copied like an integer unless it owns something.

Exchanging values that own

vasm_swap.pk

use("vasm")

report(m) := {
    es := vasm_errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Image { header: own [u8], body: own [u8], entry: u64 }
    fn replace_body(img: Image, n: i64) -> Image {
        var b: own [u8]
        b = alloc(n, 0)
        img.body = b
        return img
    }
    fn take_header[r](parts: &mut r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = alloc(1, 9)
        a = swap(parts[0], a)
        return a
    }
    fn read_header[r](parts: &r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = parts[0]
        return a
    }
|}
report(m)

Output

line 25: the elements of parts own values and are not copied; use swap(parts[i], v)

A value that owns something cannot be copied out of an array element or the target of a pointer, because it would have two owners. swap(p[i], v) puts v in and returns the previous value. Writing a field of a struct register, as in img.body = b, releases the previous value and puts the new one in.

All the rules are in vasm in the reference.
