The vasm checker

planck/lib/vasm/check.pk is a library written in planck that reads and checks vasm, a low-level language (use("vasm/check")). The rules of the language are in vasm in the reference. By type rules alone it detects violations of ownership and borrowing, use after release and reads of uninitialized values. A value is either moved or copied, and a value that has own can be moved or freed at most once in total. A borrow is valid only inside its region.

Entry points and syntax

vasm{| ... |} is read as Module{[Struct{...}, Fn{...}, ...]}. The syntax is declared with ::= as rules of the packrat parser, and statements become M-expressions such as Decl{...} and IfS{...}. Every result carries a position Pos{d}, the number of characters from the next token to the end of the text.

planck/lib/vasm/check.pk: vasm_stmt ::= at:vasm_here 'var

vasm_stmt ::= at:vasm_here 'var x:vasm_name ":" t:vasm_type { Decl{at, x, t} }
    | at:vasm_here 'if "(" c:vasm_cond ")" a:vasm_block 'else b:vasm_block { IfS{at, c, a, b} }
    | at:vasm_here 'if "(" c:vasm_cond ")" a:vasm_block { IfS{at, c, a, []} }
    | at:vasm_here 'while "(" c:vasm_cond ")" a:vasm_block { WhileS{at, c, a} }
    | at:vasm_here 'region r:vasm_name a:vasm_block { RegionS{at, r, a} }
    | at:vasm_here 'return vasm_same a:vasm_operand { Ret{at, a} }
    | at:vasm_here 'return { Ret{at, none} }
    | at:vasm_here "*" p:vasm_name "=" a:vasm_operand { Store{at, p, a} }
    | at:vasm_here p:vasm_name "[" i:vasm_operand "]" "=" a:vasm_operand { IStore{at, p, i, a} }
    | at:vasm_here p:vasm_name "." f:vasm_name "=" a:vasm_operand { FStore{at, p, f, a} }
    | at:vasm_here x:vasm_name "=" r:vasm_rhs { Assign{at, x, r} }
    | at:vasm_here f:vasm_name "(" args:vasm_args? ")" { CallS{at, f, vasm_opt_list(args)} }
  • vasm_check.check(m) returns a list of VasmErr{kind, Pos{d}, msg}. A function reports only its first error.
  • errors(file, m) turns positions into lines and columns, and report(file, m) writes file:line:column: message.

Types and the state of the checker

Types are also M-expressions: integers are I8{} to U64{}, a box is Own{u}, a borrow is Ref{'shr or 'mut, region, u}, an array is Arr{u} and a struct is Named{s}. static is a region that outlives every region.

planck/lib/vasm/check.pk: Checker state

# === Checker state ===
# vasm_info: name -> Info{type, regions open at the declaration}, for the variables in scope
# a state: name -> Vs{status, loans}, and ".dead" -> true after return
#   status: 'init, 'uninit, Moved{at}, Freed{at}, Taken{at} (an own value whose contents are moved out by x = *p)
#   or Parts{fields} (a struct value whose fields that own are moved out by x = s.f); loans: [Loan{region, kind}, ...]
# regions: the regions in scope, the innermost first, ending with the regions of the function

sigs := table.new()
structs := table.new()
vasm_info := table.new()
regions := []
fn_regions := []
ret_type := Unit{}

For each function the checker keeps the state of each variable (initialized, uninitialized, moved, freed and so on) and its list of borrows in Vs{status, loans}.

The type decides between move and copy. An own, and a struct that has own, is linear and is moved on use. Integers, bool and shared borrows are copied.

Ownership

vasm_box.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn bump[r](p: &mut r i64) {
        var v: i64
        v = *p
        v = v + 1
        *p = v
    }
    fn main() -> i64 {
        var p: own i64
        var v: i64
        p = box(41)
        region r {
            var m: &mut r i64
            m = &mut r p
            bump(m)
        }
        v = *p
        free(p)
        return v
    }
|}
report(m)

Output

ok

Before a register is read, need_init looks at its state and reports an error for uninitialized, moved or freed. An own that has been freed cannot be read because call marks the argument of free as freed. A call that passes &mut to a borrow parameter is treated as borrowing again for the duration of the call, and does not move.

vasm_drop.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Buf { data: own [u8], size: i64 }
    fn at_block_end(c: bool) {
        if (c) {
            var q: own i64
            q = box(2)
        }
    }
    fn overwrite() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        p = box(2)
        box(3)
        v = *p
        return v
    }
    fn free_with_contents(b: own Buf) {
        free(b)
    }
|}
report(m)

Output

ok

A value that has own is released automatically where it loses its owner: at the end of a block or function, on overwrite, or as a discarded result. The release is a runtime action and is not checked.

Borrows and regions

vasm_borrow.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn two_shared() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            var b: &r i64
            a = &r p
            b = &r p
            var w: i64
            v = *a
            w = *b
            v = v + w
        }
        free(p)
        return v
    }
    fn mut_and_shared() {
        var p: own i64
        p = box(1)
        region r {
            var a: &mut r i64
            var b: &r i64
            a = &mut r p
            b = &r p
        }
    }
    fn overwrite_while_borrowed() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            a = &r p
            p = box(2)
            v = *a
        }
        return v
    }
|}
report(m)

Output

line 34: p is borrowed as mut and cannot be borrowed
line 44: p is borrowed and cannot be assigned

A borrowed variable gets a Loan{region, kind}. While it is borrowed, a move, a release or an assignment is an error from no_loans. Any number of shared borrows can coexist, but only one &mut, and reads and writes during it are also errors. At the end of region r { ... } the loans of r disappear.

vasm_escape.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn inner_in_outer() {
        var p: own i64
        p = box(1)
        region outer {
            var keep: &outer i64
            region inner {
                var a: &inner i64
                a = &inner p
                keep = a
            }
        }
    }
    fn outer_in_inner() {
        var p: own i64
        p = box(1)
        region outer {
            var a: &outer i64
            a = &outer p
            region inner {
                var b: &inner i64
                b = a
            }
        }
    }
|}
report(m)

Output

line 18: a borrow of region inner is put in keep, which lives outside the region

outlives(q, r) decides which region is outside. q outlives r when it is the same as r or static, or when r is a region opened in the function and q was opened before it. A borrow of an outer region can be used as a borrow of an inner one, but not the reverse.

planck/lib/vasm/check.pk: q outlives r

# q outlives r: q is r, or q is static (the region of the string literals, which lasts to the end of the program),
# or r is a region opened in the function and q was opened before it. The regions of the function are not ordered.
outlives(q, r) := q == r or q == 'static or (r != 'static and not member(r, fn_regions) and list.index_of(q, regions) > list.index_of(r, regions))

Branches and loops

An if checks both branches on copies of the state, and merge joins them. A value moved or released in only one branch is an error, because it is not decided afterward who owns it. A while checks the body twice and verifies that a value moved in the loop is put back before the next round.

Integers, arrays and structs

vasm_ints.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn mixed(flags: u8, size: u32) -> u32 {
        var w: u32
        w = size + flags
        return w
    }
    fn widened(flags: u8, size: u32) -> u32 {
        var f: u32
        var w: u32
        f = flags as u32
        w = size + f
        return w
    }
    fn magic() -> u32 {
        var m: u32
        m = 0x7F45
        m = m << 16
        m = m | 0x4C46
        return m
    }
    fn too_big() -> u32 {
        var m: u32
        m = 0x7F454C46
        return m
    }
|}
report(m)

Output

line 12: the operands are u32 and u8
line 31: 0x7F454C46 is out of the range of a literal, -32768 to 32767; make it with << and |

Both sides of an operation have the same integer type, and as changes the width or the sign. A literal starts as an untyped Lit{n}, and its range is checked when it meets an integer type.

vasm_array.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Section { offset: u64, size: u32 }
    fn total[r](ss: &r [Section]) -> u64 {
        var i: i64
        var n: i64
        var t: u64
        var s: Section
        var o: u64
        i = 0
        t = 0
        n = len(ss)
        while (i < n) {
            s = ss[i]
            o = s.offset
            t = t + o
            i = i + 1
        }
        return t
    }
    fn build() -> u64 {
        var a: own [Section]
        var s: Section
        var t: u64
        s = Section{offset: 64, size: 16}
        a = alloc(3, s)
        s = Section{offset: 128, size: 0}
        a[1] = s
        region r {
            var view: &r [Section]
            view = &r a
            t = total(view)
        }
        free(a)
        return t
    }
|}
report(m)

Output

ok

The types of the builtin functions are in the table vasm_check.builtins. The checker checks only types and ownership, and the range of an index is checked at run time. A struct is made by writing every field once, and when it has no own it is copied like an integer.

vasm_swap.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Image { header: own [u8], body: own [u8], entry: u64 }
    fn replace_body(img: Image, n: i64) -> Image {
        var b: own [u8]
        b = alloc(n, 0)
        img.body = b
        return img
    }
    fn take_header[r](parts: &mut r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = alloc(1, 9)
        a = swap(parts[0], a)
        return a
    }
    fn read_header[r](parts: &r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = parts[0]
        return a
    }
|}
report(m)

Output

line 25: the elements of parts own values and are not copied; use swap(parts[i], v)

A value that has own cannot be copied out of an array element or from what a pointer points to. swap(p[i], v) puts v in and takes the previous value out.

Reporting errors

When the checker finds an error, fail raises the algebraic effect VasmFail{at, kind, msg} (Algebraic effects). check catches it per struct and function with handle, makes it a VasmErr and stops checking. The kinds of kind are in the table of the reference.
