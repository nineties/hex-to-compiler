vasm の検査器

planck/lib/vasm/check.pk は、低級な言語 vasm の読み込みと検査を planck で書いたライブラリです（use("vasm/check")）。言語の規則はリファレンスの vasmにあります。型の規則だけで、所有と借用の違反・解放後の使用・未初期化の読み出しを検出します。値は move かコピーのどちらかで、own を持つ値は move と解放を合わせて高々1回しかできません。借用は領域の中だけで有効です。

入口と構文

vasm{| ... |} は Module{[Struct{...}, Fn{...}, ...]} に読まれます。構文は packrat パーサの規則として ::= で宣言し、文は Decl{...}・IfS{...} などの M式になります。どの結果にも位置 Pos{d}（次の字句から本文の終わりまでの文字数）が入ります。

planck/lib/vasm/check.pk: vasm_stmt ::= at:vasm_here 'var

vasm_stmt ::= at:vasm_here 'var x:vasm_name ":" t:vasm_type { Decl{at, x, t} }
    | at:vasm_here 'if "(" c:vasm_cond ")" a:vasm_block 'else b:vasm_block { IfS{at, c, a, b} }
    | at:vasm_here 'if "(" c:vasm_cond ")" a:vasm_block { IfS{at, c, a, []} }
    | at:vasm_here 'while "(" c:vasm_cond ")" a:vasm_block { WhileS{at, c, a} }
    | at:vasm_here 'region r:vasm_name a:vasm_block { RegionS{at, r, a} }
    | at:vasm_here 'return vasm_same a:vasm_operand { Ret{at, a} }
    | at:vasm_here 'return { Ret{at, none} }
    | at:vasm_here "*" p:vasm_name "=" a:vasm_operand { Store{at, p, a} }
    | at:vasm_here p:vasm_name "[" i:vasm_operand "]" "=" a:vasm_operand { IStore{at, p, i, a} }
    | at:vasm_here p:vasm_name "." f:vasm_name "=" a:vasm_operand { FStore{at, p, f, a} }
    | at:vasm_here x:vasm_name "=" r:vasm_rhs { Assign{at, x, r} }
    | at:vasm_here f:vasm_name "(" args:vasm_args? ")" { CallS{at, f, vasm_opt_list(args)} }
  • vasm_check.check(m) は VasmErr{kind, Pos{d}, msg} のリストを返します。1つの関数の誤りは最初の1件だけです。
  • errors(file, m) は位置を行と列に直し、report(file, m) は file:行:列: メッセージ で書きます。

型と検査の状態

型も M式です。整数は I8{} から U64{}、箱は Own{u}、借用は Ref{'shr または 'mut, 領域, u}、配列は Arr{u}、構造体は Named{s} です。static はどの領域よりも長く続く領域です。

planck/lib/vasm/check.pk: Checker state

# === Checker state ===
# vasm_info: name -> Info{type, regions open at the declaration}, for the variables in scope
# a state: name -> Vs{status, loans}, and ".dead" -> true after return
#   status: 'init, 'uninit, Moved{at}, Freed{at}, Taken{at} (an own value whose contents are moved out by x = *p)
#   or Parts{fields} (a struct value whose fields that own are moved out by x = s.f); loans: [Loan{region, kind}, ...]
# regions: the regions in scope, the innermost first, ending with the regions of the function

sigs := table.new()
structs := table.new()
vasm_info := table.new()
regions := []
fn_regions := []
ret_type := Unit{}

検査器は関数ごとに、変数の状態（初期化済み・未初期化・move 済み・解放済みなど）と借用の並びを Vs{status, loans} に持ちます。

move かコピーかは型で決まります。own と、own を持つ構造体は線形で、使うと move されます。整数・bool・共有の借用はコピーされます。

所有

vasm_box.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn bump[r](p: &mut r i64) {
        var v: i64
        v = *p
        v = v + 1
        *p = v
    }
    fn main() -> i64 {
        var p: own i64
        var v: i64
        p = box(41)
        region r {
            var m: &mut r i64
            m = &mut r p
            bump(m)
        }
        v = *p
        free(p)
        return v
    }
|}
report(m)

出力

ok

レジスタを読む前に need_init が状態を見て、未初期化・move 済み・解放済みなら誤りにします。free した own が読めないのは、call が引数を解放済みにするからです。&mut を借用の引数に渡す呼び出しは、呼び出しの間だけ借り直すものとして扱い、move しません。

vasm_drop.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Buf { data: own [u8], size: i64 }
    fn at_block_end(c: bool) {
        if (c) {
            var q: own i64
            q = box(2)
        }
    }
    fn overwrite() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        p = box(2)
        box(3)
        v = *p
        return v
    }
    fn free_with_contents(b: own Buf) {
        free(b)
    }
|}
report(m)

出力

ok

own を持つ値は、ブロックや関数の終わり、上書き、捨てた結果など、持ち主がなくなる所で自動的に解放されます。解放は実行時の動作で、検査の対象ではありません。free は早く解放したいときに書きます。

借用と領域

vasm_borrow.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn two_shared() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            var b: &r i64
            a = &r p
            b = &r p
            var w: i64
            v = *a
            w = *b
            v = v + w
        }
        free(p)
        return v
    }
    fn mut_and_shared() {
        var p: own i64
        p = box(1)
        region r {
            var a: &mut r i64
            var b: &r i64
            a = &mut r p
            b = &r p
        }
    }
    fn overwrite_while_borrowed() -> i64 {
        var p: own i64
        var v: i64
        p = box(1)
        region r {
            var a: &r i64
            a = &r p
            p = box(2)
            v = *a
        }
        return v
    }
|}
report(m)

出力

line 34: p is borrowed as mut and cannot be borrowed
line 44: p is borrowed and cannot be assigned

借用した変数には Loan{region, kind} が入り、借りている間の move・解放・代入は no_loans が誤りにします。共有の借用は同時にいくつでも持てますが、&mut は1つだけで、その間の読み書きも誤りです。region r { ... } の終わりで r の貸しは消えます。

vasm_escape.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    fn inner_in_outer() {
        var p: own i64
        p = box(1)
        region outer {
            var keep: &outer i64
            region inner {
                var a: &inner i64
                a = &inner p
                keep = a
            }
        }
    }
    fn outer_in_inner() {
        var p: own i64
        p = box(1)
        region outer {
            var a: &outer i64
            a = &outer p
            region inner {
                var b: &inner i64
                b = a
            }
        }
    }
|}
report(m)

出力

line 18: a borrow of region inner is put in keep, which lives outside the region

領域の内外は outlives(q, r) が決めます。q が r と同じか static のとき、または r が関数の中で開いた領域で q がそれより前に開いたものであるとき、q は r より長く続きます。外側の領域の借用は内側の借用として使えますが、逆は誤りです。

planck/lib/vasm/check.pk: q outlives r

# q outlives r: q is r, or q is static (the region of the string literals, which lasts to the end of the program),
# or r is a region opened in the function and q was opened before it. The regions of the function are not ordered.
outlives(q, r) := q == r or q == 'static or (r != 'static and not member(r, fn_regions) and list.index_of(q, regions) > list.index_of(r, regions))

分岐とループ

if は両方の分岐を複製した状態で検査し、merge が突き合わせます。片方の分岐だけで move や解放をした値は、あとで持ち主かどうかが決まらないので誤りです。while は本体を2周分検査し、ループで move した値が次の周の前に入れ直されているかを確かめます。

整数・配列・構造体

vasm_ints.pk

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")

m := vasm{|
    fn mixed(flags: u8, size: u32) -> u32 {
        var w: u32
        w = size + flags
        return w
    }
    fn widened(flags: u8, size: u32) -> u32 {
        var f: u32
        var w: u32
        f = flags as u32
        w = size + f
        return w
    }
    fn magic() -> u32 {
        var m: u32
        m = 0x7F45
        m = m << 16
        m = m | 0x4C46
        return m
    }
    fn too_big() -> u32 {
        var m: u32
        m = 0x7F454C46
        return m
    }
|}
report(m)

出力

line 12: the operands are u32 and u8
line 31: 0x7F454C46 is out of the range of a literal, -32768 to 32767; make it with << and |

演算の両辺は同じ整数型で、幅や符号を変えるには as を使います。リテラルははじめ型のない Lit{n} で、整数の型と合わせるときに範囲を検査します。

vasm_array.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Section { offset: u64, size: u32 }
    fn total[r](ss: &r [Section]) -> u64 {
        var i: i64
        var n: i64
        var t: u64
        var s: Section
        var o: u64
        i = 0
        t = 0
        n = len(ss)
        while (i < n) {
            s = ss[i]
            o = s.offset
            t = t + o
            i = i + 1
        }
        return t
    }
    fn build() -> u64 {
        var a: own [Section]
        var s: Section
        var t: u64
        s = Section{offset: 64, size: 16}
        a = alloc(3, s)
        s = Section{offset: 128, size: 0}
        a[1] = s
        region r {
            var view: &r [Section]
            view = &r a
            t = total(view)
        }
        free(a)
        return t
    }
|}
report(m)

出力

ok

組み込みの関数の型は表 vasm_check.builtins にあります。検査するのは型と所有だけで、添字の範囲は実行時に検査します。構造体は全部のフィールドを1回ずつ書いて作り、own を持たなければ整数と同じくコピーされます。

vasm_swap.pk

use("vasm/check")

report(m) := {
    es := vasm_check.errors(argv()[0], m)
    if (es == []) put("ok\n")
    each((VasmErr{_, Loc{_, l, c}, msg}) -> put(f"line {l}: {msg}\n"), es)
}

m := vasm{|
    struct Image { header: own [u8], body: own [u8], entry: u64 }
    fn replace_body(img: Image, n: i64) -> Image {
        var b: own [u8]
        b = alloc(n, 0)
        img.body = b
        return img
    }
    fn take_header[r](parts: &mut r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = alloc(1, 9)
        a = swap(parts[0], a)
        return a
    }
    fn read_header[r](parts: &r [own [u8]]) -> own [u8] {
        var a: own [u8]
        a = parts[0]
        return a
    }
|}
report(m)

出力

line 25: the elements of parts own values and are not copied; use swap(parts[i], v)

own を持つ値は、配列の要素やポインタの指す先から読んでコピーできません。swap(p[i], v) が v を入れて前の値を取り出します。

誤りの返し方

誤りを見つけると fail が代数的エフェクト VasmFail{at, kind, msg} を起こします（代数的エフェクト）。check は構造体と関数ごとに handle でこれを受け、VasmErr にして検査を打ち切ります。kind の種類はリファレンスの表にあります。
